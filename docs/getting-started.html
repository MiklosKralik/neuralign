Getting started

Install

pip install "neuralign[examples] @ git+https://github.com/MiklosKralik/neuralign.git"

Python 3.10 or later is required. The examples extra installs Matplotlib for plotting; the model itself requires NumPy and PyTorch. For a local checkout:

uv sync --extra examples
uv run python examples/quickstart.py

The example saves examples/quickstart.png and displays the figure. Use MPLBACKEND=Agg when running without a display.

Prepare your recordings

Provide a list of floating-point arrays or tensors, each shaped neurons × time × trials. Neuron and trial counts may differ between sessions. Time bins must match, and the rank cannot exceed any session's neuron count. Choose preprocessing for your data; fit its statistics on training data when evaluating held-out observations.

Fit and inspect

from neuralign import NeuralignConfig, NeuralignModel
from neuralign.simulation import simulate_sessions

sessions, truth = simulate_sessions(seed=42)
factors = NeuralignModel(
    sessions,
    config=NeuralignConfig(rank=2, max_iter=500, seed=42, progress=False),
).fit("cpu")

session = factors.session(0)
print(factors.V.shape)                   # time, components
print(session.neuron_factors.shape)      # neurons, components
print(session.trial_factors.shape)       # trials, components
reconstructed = session.reconstructed_tensor()

Use fit("cuda") with a compatible CUDA-enabled PyTorch installation.

Complete plotting example

Download quickstart.py, or run it from the package's examples folder. Its small simulator uses the sigmoid/transient profiles and changing trial factors from the manuscript's identifiability example, with light Gaussian noise. It is a teaching example, not the full Poisson benchmark. Component order, sign and scale can differ between the generating and fitted factors.

"""Simulate, fit and plot a rank-two Neuralign model; no external data needed.

Run: python examples/quickstart.py
The figure is saved next to this script as quickstart.png.
"""

from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np

from neuralign import NeuralignConfig, NeuralignModel
from neuralign.simulation import simulate_sessions


def main():
    sessions, truth = simulate_sessions(seed=42)
    config = NeuralignConfig(
        rank=2, max_iter=500, patience=100, seed=42, progress=False
    )
    factors = NeuralignModel(sessions, config=config).fit("cpu")
    view = factors.session(0)
    reconstructed = view.reconstructed_tensor().detach().numpy()

    fig, axes = plt.subplots(2, 3, figsize=(10, 5.8), layout="constrained")
    time = truth["time"]
    for column, (V, title) in enumerate(
        (
            (truth["V"], "Generating temporal factors"),
            (factors.V.detach().numpy(), "Fitted temporal factors"),
        )
    ):
        axes[0, column].plot(time, V)
        axes[0, column].set(title=title, xlabel="Time (s)", ylabel="Factor value")
    axes[0, 2].plot(view.trial_factors.detach().numpy())
    axes[0, 2].set(
        title="Fitted trial factors · session 1",
        xlabel="Trial index",
        ylabel="Factor value",
    )

    measured = sessions[0].numpy()[:, :, 0]
    predicted = reconstructed[:, :, 0]
    limit = np.quantile(np.abs(np.stack([measured, predicted])), 0.99)
    for ax, values, title in zip(
        axes[1, :2],
        (measured, predicted),
        ("Input · first trial", "Reconstruction · first trial"),
    ):
        image = ax.imshow(
            values,
            aspect="auto",
            cmap="RdBu_r",
            vmin=-limit,
            vmax=limit,
            extent=(time[0], time[-1] + 0.05, len(values), 0),
        )
        ax.set(title=title, xlabel="Time (s)", ylabel="Neuron index")
    fig.colorbar(
        image,
        ax=axes[1, :2],
        label="Model units",
        orientation="horizontal",
        shrink=0.7,
        pad=0.15,
    )
    errors = [
        float(
            np.linalg.norm(Z.numpy() - v.reconstructed_tensor().detach().numpy())
            / np.linalg.norm(Z.numpy())
        )
        for Z, v in zip(sessions, factors)
    ]
    axes[1, 2].bar(np.arange(1, len(errors) + 1), errors, color="#536B82")
    axes[1, 2].set(
        title="Reconstruction error",
        xlabel="Session",
        ylabel="Relative Frobenius error",
        xticks=np.arange(1, len(errors) + 1),
    )
    for ax in axes.flat:
        ax.spines[["top", "right"]].set_visible(False)
    # Component order, sign and scale can differ from the generating factors.
    destination = Path(__file__).with_suffix(".png")
    fig.savefig(destination, dpi=160)
    print(f"Saved {destination}; mean relative error: {np.mean(errors):.3f}")
    plt.show()
    return factors


if __name__ == "__main__":
    main()
